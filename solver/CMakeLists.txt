cmake_minimum_required(VERSION 3.21)

# The pyTEC3D solver: the core library (libMesh, PETSc, MPI; no Python, no VTK) and its pybind11 module.
# Built by scikit-build-core (pip install ./solver, see pyproject.toml) or by plain CMake (cmake --install puts the
# module and its __init__.py in <prefix>/pytec3d_solver).

# Single source of the version: the pyproject.toml of the solver.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/pyproject.toml" PYTEC3D_VERSION_LINE REGEX "^version = \"[0-9.]+\"")
string(REGEX REPLACE "^version = \"([0-9.]+)\"" "\\1" PYTEC3D_VERSION "${PYTEC3D_VERSION_LINE}")
if(NOT PYTEC3D_VERSION)
  message(FATAL_ERROR "Could not read the version from pyproject.toml.")
endif()

project(pyTEC3D VERSION ${PYTEC3D_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON) # for clang-tidy (.clang-tidy)

# --- Dependencies ----------------------------------------------------------------
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(MPI REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PETSC QUIET IMPORTED_TARGET PETSc)
if(NOT PETSC_FOUND)
  pkg_check_modules(PETSC REQUIRED IMPORTED_TARGET petsc)
endif()

# The Python interpreter runs the registry generator at build time; the Python module (PYTEC3D_BUILD_PYTHON_MODULE)
# also needs pybind11 and the Python headers. The solver core itself depends on neither, nor on VTK.
option(PYTEC3D_BUILD_PYTHON_MODULE "Build the pybind11 module pytec3d_solver (else only the C++ core library)" ON)
option(PYTEC3D_BUILD_CPP_EXAMPLES "Build the C++ examples (examples/), which use the core without Python" OFF)
if(PYTEC3D_BUILD_PYTHON_MODULE)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
else()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()

# --- libMesh -----------------------------------------------------------------------
# LIBMESH_DIR is the install prefix (contains include/ and lib/).
set(LIBMESH_DIR "" CACHE PATH "Path to the libMesh installation prefix.")
if(NOT LIBMESH_DIR AND DEFINED ENV{LIBMESH_DIR})
  set(LIBMESH_DIR "$ENV{LIBMESH_DIR}")
endif()
if(NOT LIBMESH_DIR)
  message(FATAL_ERROR "LIBMESH_DIR must be provided (e.g., -DLIBMESH_DIR=/path/to/libmesh).")
endif()

set(LIBMESH_METHOD "$ENV{pyTEC3D_LIBMESH_METHOD}" CACHE STRING "libMesh method to link against: opt, dbg, devel")
if(NOT LIBMESH_METHOD)
  set(LIBMESH_METHOD "opt")
endif()
string(TOLOWER "${LIBMESH_METHOD}" LIBMESH_METHOD)
if(NOT LIBMESH_METHOD MATCHES "^(opt|dbg|devel)$")
  message(FATAL_ERROR "Invalid LIBMESH_METHOD='${LIBMESH_METHOD}'. Allowed: opt, dbg, devel.")
endif()
message(STATUS "Using libMesh method: ${LIBMESH_METHOD}")

# NO_DEFAULT_PATH: never silently pick up a different libMesh from the env.
find_library(LIBMESH_LIB
  NAMES mesh_${LIBMESH_METHOD}
  PATHS "${LIBMESH_DIR}/lib"
  NO_DEFAULT_PATH
  REQUIRED
)
find_library(TIMPI_LIB
  NAMES timpi_${LIBMESH_METHOD}
  PATHS "${LIBMESH_DIR}/lib"
  NO_DEFAULT_PATH
  REQUIRED
)

# The JSON registries are compiled into the module: fields, domains, parameters and laws become enums
# (registries.h), each element type a table of enums and indices with its compiled expressions and laws
# (formulations.h), generated by tools/generate_registries.py with registries.cpp. Editing any of them means rebuilding the
# module. Only the model data JSON is read at run time.
# They are the package pytec3d-element-library (element_library/ of the repository), which the Python API also reads:
# the copy of the repository when the solver is built from it, else the installed package.
set(PYTEC3D_ELEMENT_LIBRARY_IN_TREE "${CMAKE_CURRENT_SOURCE_DIR}/../element_library/pytec3d_element_library/data")
if(EXISTS "${PYTEC3D_ELEMENT_LIBRARY_IN_TREE}/element_types.json")
  set(PYTEC3D_ELEMENT_LIBRARY_DEFAULT "${PYTEC3D_ELEMENT_LIBRARY_IN_TREE}")
else()
  execute_process(COMMAND "${Python3_EXECUTABLE}" -m pytec3d_element_library
                  OUTPUT_VARIABLE PYTEC3D_ELEMENT_LIBRARY_DEFAULT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
set(PYTEC3D_RESOURCE_DIR "${PYTEC3D_ELEMENT_LIBRARY_DEFAULT}" CACHE PATH
  "Directory of the JSON element library (element_types.json, fields.json, nonlinear_laws.json, domains.json)")
if(NOT EXISTS "${PYTEC3D_RESOURCE_DIR}/element_types.json")
  message(FATAL_ERROR "The element library was not found: install pytec3d-element-library, or set PYTEC3D_RESOURCE_DIR.")
endif()
message(STATUS "Element library: ${PYTEC3D_RESOURCE_DIR}")
set(PYTEC3D_REGISTRY_JSON
  "${PYTEC3D_RESOURCE_DIR}/element_types.json"
  "${PYTEC3D_RESOURCE_DIR}/fields.json"
  "${PYTEC3D_RESOURCE_DIR}/nonlinear_laws.json"
  "${PYTEC3D_RESOURCE_DIR}/domains.json")

set(PYTEC3D_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(PYTEC3D_GENERATED_FILES
  "${PYTEC3D_GENERATED_DIR}/registries.h"
  "${PYTEC3D_GENERATED_DIR}/registries.cpp"
  "${PYTEC3D_GENERATED_DIR}/formulations.h")
add_custom_command(
  OUTPUT ${PYTEC3D_GENERATED_FILES}
  COMMAND "${CMAKE_COMMAND}" -E make_directory "${PYTEC3D_GENERATED_DIR}"
  COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_registries.py"
          --resources "${PYTEC3D_RESOURCE_DIR}" --output "${PYTEC3D_GENERATED_DIR}"
  DEPENDS ${PYTEC3D_REGISTRY_JSON} "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_registries.py"
  COMMENT "Compiling the JSON registries and element types"
  VERBATIM)
list(APPEND SOURCES ${PYTEC3D_GENERATED_FILES})

# The git commit of the sources, for the banner of the solver log (git_commit.h, refreshed at every build).
add_custom_target(pytec3d_git_commit
  COMMAND "${CMAKE_COMMAND}" -E make_directory "${PYTEC3D_GENERATED_DIR}"
  COMMAND "${CMAKE_COMMAND}" -DREPO=${pyTEC3D_REPO_ROOT} -DOUT=${PYTEC3D_GENERATED_DIR}/git_commit.h
          -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/git_commit.cmake"
  BYPRODUCTS "${PYTEC3D_GENERATED_DIR}/git_commit.h"
  COMMENT "Reading the git commit"
  VERBATIM)

# --- Runtime library search path (RPATH) ----------------------------------------
# The module is copied as-is by pip (no relinking), so the RPATH must be baked in at build time. Absolute paths:
# libMesh/TIMPI install dirs + the env's lib/ (PETSc, MPI, ...). Works on the machine that built libMesh. For
# relocatable wheels, bundle the libs and use auditwheel instead.
get_filename_component(LIBMESH_LIB_DIR "${LIBMESH_LIB}" DIRECTORY)
get_filename_component(TIMPI_LIB_DIR   "${TIMPI_LIB}"   DIRECTORY)

set(PYTEC3D_RUNTIME_PREFIX "$ENV{CONDA_PREFIX}" CACHE PATH
  "Env prefix whose lib/ holds PETSc/MPI at runtime")

set(PYTEC3D_RPATH "${LIBMESH_LIB_DIR}" "${TIMPI_LIB_DIR}")
if(PYTEC3D_RUNTIME_PREFIX)
  list(APPEND PYTEC3D_RPATH "${PYTEC3D_RUNTIME_PREFIX}/lib")
endif()
list(REMOVE_DUPLICATES PYTEC3D_RPATH)
message(STATUS "Module RPATH: ${PYTEC3D_RPATH}")

# --- Solver core: plain C++ (libMesh, PETSc, MPI), no Python, no VTK -------------------------------------------

  "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${PYTEC3D_GENERATED_DIR}"

# Third-party headers: SYSTEM, so that the warnings below are about this project only.

  "${LIBMESH_DIR}/include"

  list(APPEND pyTEC3D_WARNING_OPTIONS $<$<CONFIG:Debug>:-Werror>)

# DT_RPATH (not DT_RUNPATH) so the search path also applies to transitive deps. Windows has no RPATH: the DLLs are
# next to the module or in the Library\bin of the conda environment (see python/pytec3d_solver/__init__.py).
function(pytec3d_set_rpath target)
  if(UNIX)
    set_target_properties(${target} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${PYTEC3D_RPATH}")
    if(NOT APPLE)
      target_link_options(${target} PRIVATE "LINKER:--disable-new-dtags")
    endif()
  endif()
endfunction()

# MinGW (the Windows build, MSYS2): the GCC and C++ runtimes are linked statically; the DLLs of MSYS2 the module still
# needs are shipped next to it (ci/windows/build.sh).
# Libraries to add to the link (e.g. what a static libMesh or PETSc of another configuration needs).
set(PYTEC3D_EXTRA_LIBS "" CACHE STRING "Extra libraries linked to the solver core")
if(PYTEC3D_EXTRA_LIBS)
  target_link_libraries(pytec3d_core PUBLIC ${PYTEC3D_EXTRA_LIBS})
endif()

if(MINGW)
  # Static PETSc (the Windows build): its private dependencies (MUMPS, ScaLAPACK, METIS, BLAS, Fortran runtime) too.
  target_link_libraries(pytec3d_core PUBLIC ${PETSC_STATIC_LDFLAGS} ws2_32)
  set(PYTEC3D_STATIC_RUNTIME_OPTIONS -static-libgcc -static-libstdc++)
endif()

# --- Python module: the pybind11 bindings of the core -------------------------------------------------------------
if(PYTEC3D_BUILD_PYTHON_MODULE)
  pybind11_add_module(pytec3d_solver MODULE "${SOURCE_DIR}/python_bindings.cpp")
  target_link_libraries(pytec3d_solver PRIVATE pytec3d_core)
  target_compile_options(pytec3d_solver PRIVATE ${pyTEC3D_WARNING_OPTIONS})
  if(MINGW)
    target_link_options(pytec3d_solver PRIVATE ${PYTEC3D_STATIC_RUNTIME_OPTIONS})
  endif()
  # The extension suffix of the Python it is built for (pytec3d_solver.cpython-312-x86_64-linux-gnu.so,
  # pytec3d_solver.cp312-win_amd64.pyd): one file per Python version.
  pytec3d_set_rpath(pytec3d_solver)

  if(SKBUILD)
    # scikit-build-core: into the wheel, next to python/pytec3d_solver/__init__.py.
    install(TARGETS pytec3d_solver LIBRARY DESTINATION pytec3d_solver COMPONENT python)
  else()
    install(TARGETS pytec3d_solver LIBRARY DESTINATION pytec3d_solver RUNTIME DESTINATION pytec3d_solver
            COMPONENT python)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/python/pytec3d_solver/__init__.py" DESTINATION pytec3d_solver
            COMPONENT python)
  endif()
endif()

# --- C++ examples: programs that drive the core directly ------------------------------------------------------------

  file(GLOB PYTEC3D_CPP_EXAMPLES "${CMAKE_CURRENT_SOURCE_DIR}/examples/*.cpp")

    if(MINGW)
      target_link_options(${example_name} PRIVATE ${PYTEC3D_STATIC_RUNTIME_OPTIONS})
    endif()
    pytec3d_set_rpath(${example_name})
  endforeach()
endif()
